cmake_minimum_required(VERSION 3.20)
project(kvlog LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(Threads REQUIRED)
add_library(kvlog INTERFACE)
target_include_directories(kvlog INTERFACE include)
target_link_libraries(kvlog INTERFACE Threads::Threads)
if(WIN32)
  target_link_libraries(kvlog INTERFACE ws2_32)
endif()
if(MSVC)
  target_compile_options(kvlog INTERFACE /W4 /permissive-)
else()
  target_compile_options(kvlog INTERFACE -Wall -Wextra -Wpedantic)
endif()

add_executable(kvlog-server src/server.cpp)
target_link_libraries(kvlog-server PRIVATE kvlog)
# Same server with plain fsync instead of fdatasync on the log's hot path, for the benchmark's A/B row.
add_executable(kvlog-server-fsync src/server.cpp)
target_link_libraries(kvlog-server-fsync PRIVATE kvlog)
target_compile_definitions(kvlog-server-fsync PRIVATE KVLOG_SYNC_WITH_FSYNC)
add_executable(kvlog-bench src/bench.cpp)
target_link_libraries(kvlog-bench PRIVATE kvlog)

enable_testing()
foreach(t test_resp test_log test_store)
  add_executable(${t} tests/${t}.cpp)
  target_link_libraries(${t} PRIVATE kvlog)
  add_test(NAME ${t} COMMAND ${t} WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
endforeach()

# Mutation check, enforced: the same store test compiled with the write-ordering bug deliberately reintroduced
# (log appended after releasing the shard lock) must FAIL. If it ever passes, test_store has stopped detecting it.
add_executable(test_store_broken tests/test_store.cpp)
target_link_libraries(test_store_broken PRIVATE kvlog)
target_compile_definitions(test_store_broken PRIVATE KVLOG_BROKEN_ORDERING ORDERING_ONLY)
add_test(NAME test_store_broken_must_fail COMMAND test_store_broken WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
set_tests_properties(test_store_broken_must_fail PROPERTIES WILL_FAIL TRUE)
